```dataview
TABLE WITHOUT ID 
	key AS "unresolved link", 
	rows.file.link AS "referencing file"
FROM "Attachments"
FLATTEN file.outlinks as outlinks
WHERE !(outlinks.file) AND !(contains(meta(outlinks).path, "/"))
GROUP BY outlinks
```
